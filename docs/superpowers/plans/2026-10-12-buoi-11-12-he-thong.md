## Bảng cắm thiết bị (Grove Base Hat) — user cắm theo bảng này ở Task 4

> Tắt Pi trước khi cắm (`sudo poweroff`, chờ đèn xanh tắt, rút nguồn). Cổng D có 2 chân; module dùng chân đầu = số GPIO ghi trên cổng. LED Grove: chân dài của bóng LED cắm vào lỗ "+" trên module.

**Pi 1 — pi4-tdbao (Slave 1 + Gateway)**

| # | Thiết bị | Cổng Grove | Chân trong code |
|---|---|---|---|
| 1 | DHT11 (nhiệt độ + độ ẩm) | **D5** | `DHT("11", 5)` |
| 2 | Cảm biến khoảng cách (Ultrasonic Ranger) | **D16** | `GroveUltrasonicRanger(16)` |
| 3 | LED 1 | **D18** | `LED(18)` |
| 4 | LED 2 | **D22** | `LED(22)` |
| 5 | Biến trở (Rotary Angle Sensor) | **A0** | `ADC(0x08).read_voltage(0)` |

Để trống: D24, D26, A2, A4, A6, I2C, PWM, UART.

**Pi 2 — Pi nhóm 3 (Slave 2)**

| # | Thiết bị | Cổng Grove | Chân trong code |
|---|---|---|---|
| 1 | LED 3 | **D18** | `LED(18)` |
| 2 | Buzzer | **D24** | `Buzzer(24)` |
| 3 | Relay | **D26** | `OutputDevice(26)` |
| 4 | Cảm biến ánh sáng (Light Sensor) | **A0** | `ADC(0x08).read(0)` |
| 5 | LCD 16x2 (JHD1802) | **I2C** (1 trong 3 cổng I2C, cổng nào cũng được) | `JHD1802()` địa chỉ `0x3E` |
| 6 | USB LoRa (CH340) | cổng **USB** của Pi (đang cắm sẵn) | `/dev/ttyUSB0` |

Để trống: D5, D16, D22, A2, A4, A6, PWM, UART.

## Sơ đồ & giao thức

```
 Web (trình duyệt) --HTTP :8000--> LAPTOP master (server.py + EMQX :1883 + MongoDB Atlas)
                                        |  MQTT
                                        v
                 pi4-tdbao: slave1.py (cảm biến + LED1/2)  +  gateway.py (MQTT <-> UDP)
                                                                  | UDP lệnh -> :5005
                                                                  | UDP dữ liệu <- :5006
                                                                  v
                 Pi nhóm 3: slave2.py (LED3/Buzzer/Relay/ánh sáng/LCD) --USB LoRa--> IoT node 3 (Light reg 9, Alarm reg 8)
```

| Topic MQTT | Ai publish | Ai subscribe | Nội dung (JSON) |
|---|---|---|---|
| `buoi11/slave1/trung_binh` | slave1 (20 s/lần) | master | `{"nhiet_do","do_am","bien_tro","khoang_cach","so_mau"}` |
| `buoi11/slave1/trang_thai` | slave1 (1 s/lần) | master | `{"led1","led2"}` |
| `buoi11/slave2/trung_binh` | gateway (20 s/lần) | master | `{"anh_sang","so_mau"}` |
| `buoi11/slave2/trang_thai` | gateway (1 s/lần) | master | `{"led3","buzzer","relay","light","alarm","lora_ok"}` |
| `buoi11/dieu_khien` (retain) | master (khi đổi) | slave1, gateway | `{"led1","led2","led3","buzzer","relay","light","alarm"}` 0/1 |

UDP: gateway → Slave 2 cổng **5005**: `{"led3","buzzer","relay","light","alarm"}` (khi có lệnh mới + nhắc lại mỗi 1 s). Slave 2 → gateway cổng **5006**: `{"loai":"trang_thai",…}` mỗi 1 s, `{"loai":"trung_binh","anh_sang","so_mau"}` mỗi 20 s.

MongoDB `iot_buoi11`: collection `trang_thai` (mỗi gói 1 s = 1 document), `trung_binh` (mỗi gói 20 s = 1 document), `cai_dat` (mỗi lần Set = 1 document; khởi động lại master thì nạp bản mới nhất). Mọi document có `nguon` và `thoi_gian` do master gán.

```python
"""
Buoi 11+12 - cua so trung binh dung chung cho 2 slave (copy file nay len ca 2 Pi).

Moi giay goi them() 1 lan voi 1 mau {ten_cam_bien: gia_tri hoac None}. Du
SO_MAU lan (20 lan = 20 giay) thi tra ve trung binh tung cam bien de gui 1
goi tin ve master - giong buoi 5 gom 20 giay roi moi gui ThingSpeak. Lan doc
loi (None) khong tinh vao trung binh; ca 20 lan deu loi thi ra None.
"""

SO_MAU = 20


class CuaSoTrungBinh:
    def __init__(self, so_mau=SO_MAU, chu_so=2):
        self.so_mau = so_mau
        self.chu_so = chu_so
        self._bat_dau_lai()

    def _bat_dau_lai(self):
        self._dem = 0
        self._tong = {}
        self._so_lan = {}

    def them(self, mau):
        self._dem += 1
        for ten, gia_tri in mau.items():
            self._tong.setdefault(ten, 0.0)
            self._so_lan.setdefault(ten, 0)
            if gia_tri is not None:
                self._tong[ten] += float(gia_tri)
                self._so_lan[ten] += 1
        if self._dem < self.so_mau:
            return None
        ket_qua = {
            ten: round(self._tong[ten] / self._so_lan[ten], self.chu_so) if self._so_lan[ten] else None
            for ten in self._tong
        }
        ket_qua["so_mau"] = self._dem
        self._bat_dau_lai()
        return ket_qua
```

```python
"""
Buoi 11+12 - LOGIC cua MASTER, tach rieng khoi MQTT/HTTP/Mongo de test duoc.

HeThong giu: du lieu doc ve tu 2 slave, nut bam tren web (thu_cong), lich
Light/Alarm Light, bang nguong. Moi khi co thay doi thi tinh lai lenh cho 7
dau ra va publish (chi khi khac lan truoc) len buoi11/dieu_khien.

"Set" tren web = THAY TOAN BO cai dat cu bang ban moi (khong tron voi ban
cu, o trong / bo tick deu co nghia) -> khong bao gio "ket o lan luu truoc".
"""

import copy
import json
import re
import threading
from datetime import datetime, timedelta, timezone

VN = timezone(timedelta(hours=7))

SLAVE1_TRUNG_BINH = "buoi11/slave1/trung_binh"
SLAVE1_TRANG_THAI = "buoi11/slave1/trang_thai"
SLAVE2_TRUNG_BINH = "buoi11/slave2/trung_binh"
SLAVE2_TRANG_THAI = "buoi11/slave2/trang_thai"
DIEU_KHIEN = "buoi11/dieu_khien"
TOPIC_NHAN = {
    SLAVE1_TRUNG_BINH: ("slave1", "trung_binh"),
    SLAVE1_TRANG_THAI: ("slave1", "trang_thai"),
    SLAVE2_TRUNG_BINH: ("slave2", "trung_binh"),
    SLAVE2_TRANG_THAI: ("slave2", "trang_thai"),
}

DAU_RA = ("led1", "led2", "led3", "buzzer", "relay", "light", "alarm")
DEN_NODE = ("light", "alarm")
LED_NGUONG = ("led1", "led2", "led3")
MAT_KET_NOI_GIAY = 5  # qua 5 giay khong nhan trang_thai -> web bao mat ket noi
_GIO = re.compile(r"([01]\d|2[0-3]):[0-5]\d")


class LoiDuLieu(ValueError):
    """Du lieu gui len sai -> web nhan HTTP 400, cai dat cu giu nguyen."""


def trong_khung(bat, tat, bay_gio):
    if bat <= tat:
        return bat <= bay_gio < tat
    return bay_gio >= bat or bay_gio < tat  # khung qua dem, vd 18:00 -> 06:00


def kiem_tra_lich(du_lieu):
    if not isinstance(du_lieu, dict) or du_lieu.get("che_do") not in ("tay", "tu_dong"):
        raise LoiDuLieu("che_do phai la 'tay' hoac 'tu_dong'")
    lich = {"che_do": du_lieu["che_do"]}
    for den in DEN_NODE:
        khung = du_lieu.get(den)
        if not isinstance(khung, dict):
            raise LoiDuLieu(f"thieu gio bat/tat cua {den}")
        for moc in ("bat", "tat"):
            if not isinstance(khung.get(moc), str) or not _GIO.fullmatch(khung[moc]):
                raise LoiDuLieu(f"gio {moc} cua {den} phai dang HH:MM")
        lich[den] = {"bat": khung["bat"], "tat": khung["tat"]}
    return lich


def _so_hoac_none(gia_tri, ten):
    if gia_tri is None or (isinstance(gia_tri, str) and not gia_tri.strip()):
        return None  # o trong = bo dieu kien nay
    try:
        so = float(gia_tri)
    except (TypeError, ValueError):
        raise LoiDuLieu(f"{ten} phai la so") from None
    if not 0 <= so <= 100:
        raise LoiDuLieu(f"{ten} phai trong khoang 0-100")
    return so


def kiem_tra_nguong(du_lieu):
    if not isinstance(du_lieu, dict) or not isinstance(du_lieu.get("den", {}), dict):
        raise LoiDuLieu("sai dinh dang bang nguong")
    den = du_lieu.get("den", {})
    return {
        "nhiet_do": _so_hoac_none(du_lieu.get("nhiet_do"), "nhiet_do"),
        "do_am": _so_hoac_none(du_lieu.get("do_am"), "do_am"),
        "den": {ten: den.get(ten) is True for ten in LED_NGUONG},  # khong tick / thieu = khong chon
    }


def vuot_nguong(nguong, trung_binh):
    for ten in ("nhiet_do", "do_am"):
        muc, do_duoc = nguong[ten], trung_binh.get(ten)
        if muc is not None and isinstance(do_duoc, (int, float)) and do_duoc >= muc:
            return True
    return False


class HeThong:
    def __init__(self, publish, luu_db, bay_gio=lambda: datetime.now(VN)):
        self._publish = publish  # publish(topic, chuoi_json)
        self._luu_db = luu_db  # luu_db(ten_collection, document)
        self._bay_gio = bay_gio
        self._khoa = threading.RLock()
        self._da_gui = None
        self.thu_cong = dict.fromkeys(DAU_RA, 0)
        self.lich = {
            "che_do": "tay",
            "light": {"bat": "18:00", "tat": "06:00"},
            "alarm": {"bat": "22:00", "tat": "05:00"},
            "luc": None,
        }
        self.nguong = {"nhiet_do": None, "do_am": None, "den": dict.fromkeys(LED_NGUONG, False), "luc": None}
        self.slave = {
            ten: {"trung_binh": {}, "trang_thai": {}, "luc_trung_binh": None, "luc_trang_thai": None}
            for ten in ("slave1", "slave2")
        }
        self.dieu_khien = dict.fromkeys(DAU_RA, 0)

    # --- du lieu tu slave -------------------------------------------------
    def nhan_mqtt(self, topic, payload):
        if topic not in TOPIC_NHAN:
            return
        du_lieu = json.loads(payload)
        if not isinstance(du_lieu, dict):
            raise LoiDuLieu("payload phai la JSON object")
        nguon, loai = TOPIC_NHAN[topic]
        luc = self._bay_gio()  # gio master - dong ho Pi nhom 3 bi sai
        with self._khoa:
            self.slave[nguon][loai] = du_lieu
            self.slave[nguon]["luc_" + loai] = luc
        self._luu_db(loai, {**du_lieu, "nguon": nguon, "thoi_gian": luc})
        if topic == SLAVE1_TRUNG_BINH:
            self.tinh_lai()  # nhiet do/do am moi -> xet lai nguong

    # --- tu web -----------------------------------------------------------
    def bam_nut(self, ten, gia_tri):
        if ten not in DAU_RA:
            raise LoiDuLieu(f"khong co thiet bi '{ten}'")
        if gia_tri not in (0, 1):
            raise LoiDuLieu("gia_tri phai la 0 hoac 1")
        with self._khoa:
            self.thu_cong[ten] = int(gia_tri)
        return self.tinh_lai()

    def luu_lich(self, du_lieu):
        moi = kiem_tra_lich(du_lieu)  # sai thi raise truoc khi dung vao cai dat cu
        return self._luu_cai_dat("lich", moi)

    def luu_nguong(self, du_lieu):
        moi = kiem_tra_nguong(du_lieu)
        return self._luu_cai_dat("nguong", moi)

    def _luu_cai_dat(self, loai, moi):
        luc = self._bay_gio()
        moi["luc"] = luc.strftime("%H:%M:%S")
        with self._khoa:
            setattr(self, loai, moi)
        self._luu_db("cai_dat", {"loai": loai, **copy.deepcopy(moi), "thoi_gian": luc})
        self.tinh_lai()
        return copy.deepcopy(moi)

    def nap_cai_dat(self, lich, nguong):
        """Lay lai lan Set cuoi cung (doc tu Database luc khoi dong). Hong thi bo qua."""
        with self._khoa:
            try:
                self.lich = {**kiem_tra_lich(lich), "luc": lich.get("luc")}
            except LoiDuLieu:
                pass
            try:
                self.nguong = {**kiem_tra_nguong(nguong), "luc": nguong.get("luc")}
            except LoiDuLieu:
                pass

    # --- tinh lenh dieu khien ------------------------------------------------
    def tinh_dieu_khien(self):
        with self._khoa:
            ra = dict(self.thu_cong)
            if self.lich["che_do"] == "tu_dong":
                gio = self._bay_gio().strftime("%H:%M")
                for den in DEN_NODE:
                    ra[den] = int(trong_khung(self.lich[den]["bat"], self.lich[den]["tat"], gio))
            if vuot_nguong(self.nguong, self.slave["slave1"]["trung_binh"]):
                for ten, chon in self.nguong["den"].items():
                    if chon:
                        ra[ten] = 1
            return ra

    def tinh_lai(self):
        with self._khoa:
            ra = self.tinh_dieu_khien()
            self.dieu_khien = ra
            chuoi = json.dumps(ra, sort_keys=True)
            if chuoi != self._da_gui:
                self._da_gui = chuoi
                self._publish(DIEU_KHIEN, chuoi)
            return dict(ra)

    def gui_lai(self):
        """Publish lai du khong doi (vd vua ket noi lai broker)."""
        with self._khoa:
            self._da_gui = None
            return self.tinh_lai()

    # --- cho web ------------------------------------------------------------
    def anh_chup(self):
        bay_gio = self._bay_gio()
        with self._khoa:
            ket_qua = {}
            for ten, s in self.slave.items():
                luc_tt = s["luc_trang_thai"]
                ket_qua[ten] = {
                    "trung_binh": dict(s["trung_binh"]),
                    "trang_thai": dict(s["trang_thai"]),
                    "luc_trung_binh": s["luc_trung_binh"].strftime("%H:%M:%S") if s["luc_trung_binh"] else None,
                    "ket_noi": luc_tt is not None and (bay_gio - luc_tt).total_seconds() <= MAT_KET_NOI_GIAY,
                }
            ket_qua.update(
                dieu_khien=dict(self.dieu_khien),
                thu_cong=dict(self.thu_cong),
                lich=copy.deepcopy(self.lich),
                nguong=copy.deepcopy(self.nguong),
                vuot_nguong=vuot_nguong(self.nguong, self.slave["slave1"]["trung_binh"]),
                gio=bay_gio.strftime("%H:%M:%S"),
            )
            return ket_qua
```

```python
"""
Buoi 11+12 - MASTER (laptop): MQTT (EMQX) + web HTTP + MongoDB Atlas.

def _doc_file_env():
    gia_tri = {}
    file_env = Path(__file__).with_name(".env")
    if file_env.exists():
        for dong in file_env.read_text(encoding="utf-8").splitlines():
            dong = dong.strip()
            if dong and not dong.startswith("#") and "=" in dong:
                ten, gt = dong.split("=", 1)
                gia_tri[ten.strip()] = gt.strip().strip('"').strip("'")
    return gia_tri


_ENV = _doc_file_env()

    print("Da ket noi MQTT:", reason_code)
    client.subscribe([(topic, 1) for topic in TOPIC_NHAN])
    he_thong.gui_lai()

    except Exception as loi:  # paho 2: loi lot ra callback se lam chet luong MQTT

    try:
        lich = db.cai_dat.find_one({"loai": "lich"}, sort=[("_id", DESCENDING)])
        nguong = db.cai_dat.find_one({"loai": "nguong"}, sort=[("_id", DESCENDING)])
        he_thong.nap_cai_dat(lich, nguong)
        print("Da nap cai dat lan Set cuoi tu Database")
    except Exception as loi:
        print("[DB] Khong nap duoc cai dat cu (dung mac dinh):", loi)


class Handler(BaseHTTPRequestHandler):
    def tra(self, ma, noi_dung, kieu="application/json"):
        if isinstance(noi_dung, str):
            noi_dung = noi_dung.encode()
        self.send_response(ma)
        self.send_header("Content-Type", kieu)
        self.send_header("Content-Length", str(len(noi_dung)))
        self.send_header("Cache-Control", "no-store")
        self.end_headers()
        self.wfile.write(noi_dung)

    def do_GET(self):
        if self.path == "/api/trang_thai":
            self.tra(200, json.dumps(he_thong.anh_chup()))
        elif self.path in ("/", "/index.html"):
            self.tra(200, TRANG_WEB, "text/html; charset=utf-8")
        else:
            self.tra(404, '{"loi": "khong co trang nay"}')

    def do_POST(self):
        try:
            than = self.rfile.read(int(self.headers.get("Content-Length") or 0))
            du_lieu = json.loads(than or b"null")
            if self.path == "/api/nut":
                if not isinstance(du_lieu, dict):
                    raise LoiDuLieu("sai dinh dang")
                ket_qua = he_thong.bam_nut(du_lieu.get("ten"), du_lieu.get("gia_tri"))
            elif self.path == "/api/lich":
                ket_qua = he_thong.luu_lich(du_lieu)
            elif self.path == "/api/nguong":
                ket_qua = he_thong.luu_nguong(du_lieu)
            else:
                return self.tra(404, '{"loi": "khong co API nay"}')
        except ValueError as loi:  # LoiDuLieu + JSON hong
            return self.tra(400, json.dumps({"loi": str(loi)}))
        self.tra(200, json.dumps(ket_qua))

    def log_message(self, *args):
        pass


if __name__ == "__main__":
    nap_cai_dat_cu()
    threading.Thread(target=vong_db, daemon=True).start()
    client.on_connect = on_connect
    client.on_message = on_message
    client.connect_async(MQTT_HOST, MQTT_PORT)
    client.loop_start()
    threading.Thread(target=vong_1_giay, daemon=True).start()
    print(f"Master: web http://localhost:{HTTP_PORT} | MQTT {MQTT_HOST}:{MQTT_PORT} | DB {MONGODB_DB}")
    ThreadingHTTPServer(("0.0.0.0", HTTP_PORT), Handler).serve_forever()
```

- [ ] **Step 1: Viết `slave1/slave1.py`**

```python
"""
Buoi 11+12 - SLAVE 1 (pi4-tdbao): DHT11 + bien tro + khoang cach + LED1/LED2.

Moi 1 giay: doc 3 cam bien vao cua so trung binh; doc trang thai LED1/LED2
va publish ngay len buoi11/slave1/trang_thai. Du 20 mau -> publish 1 goi
trung binh len buoi11/slave1/trung_binh. Subscribe buoi11/dieu_khien
(master publish, retain) de bat/tat LED1, LED2.

Cam day (Grove Base Hat):
    DHT11        -> D5   (GPIO5)
    Khoang cach  -> D16  (GPIO16)
    LED 1        -> D18  (GPIO18)
    LED 2        -> D22  (GPIO22)
    Bien tro     -> A0   (ADC kenh 0)

Chay (cung thu muc voi trung_binh.py):
    IOT_MQTT_HOST=192.168.44.179 python3 slave1.py
"""

import json
import os
import time

import paho.mqtt.client as mqtt
from gpiozero import LED
from grove.adc import ADC
from grove.grove_ultrasonic_ranger import GroveUltrasonicRanger
from seeed_dht import DHT

from trung_binh import CuaSoTrungBinh

MQTT_HOST = os.environ.get("IOT_MQTT_HOST", "192.168.44.179")
MQTT_PORT = int(os.environ.get("IOT_MQTT_PORT", "1883"))
TRUNG_BINH = "buoi11/slave1/trung_binh"
TRANG_THAI = "buoi11/slave1/trang_thai"
DIEU_KHIEN = "buoi11/dieu_khien"

dht = DHT("11", 5)
sieu_am = GroveUltrasonicRanger(16)
adc = ADC(0x08)
led = {"led1": LED(18), "led2": LED(22)}


def doc_dht():
    try:
        do_am, nhiet_do = dht.read()
        do_am, nhiet_do = float(do_am), float(nhiet_do)
    except Exception as loi:
        print("Loi doc DHT11:", loi)
        return None, None
    # DHT11 do 0-50 C, 20-95 %; ngoai khoang (vd 0/0 khi chua doc duoc lan nao) = doc hong
    if not (0 <= nhiet_do <= 50 and 20 <= do_am <= 95):
        return None, None
    return nhiet_do, do_am


def doc_bien_tro():
    try:
        return round(adc.read_voltage(0) / 1000, 3)  # mV -> V (0-3.3)
    except Exception as loi:
        print("Loi doc bien tro:", loi)
        return None


def doc_khoang_cach():
    # KHONG goi get_distance(): ham do lap VO HAN khi khong co tieng vong (rut
    # cam bien / vat qua xa). _get_distance() do 1 lan, het gio thi tra None.
    for _ in range(3):
        try:
            kc = sieu_am._get_distance()
        except Exception:
            kc = None
        if kc is not None and 2 <= kc <= 400:
            return round(kc, 1)
    return None


def on_connect(client, userdata, connect_flags, reason_code, properties=None):
    print("Da ket noi MQTT:", reason_code)
    client.subscribe(DIEU_KHIEN, qos=1)


def on_message(client, userdata, msg):
    try:
        lenh = json.loads(msg.payload)
        for ten, bong in led.items():
            if ten in lenh:
                bong.on() if lenh[ten] else bong.off()
    except Exception as loi:
        print("Lenh hong:", loi, msg.payload)


def main():
    client = mqtt.Client(mqtt.CallbackAPIVersion.VERSION2, client_id="slave1-buoi11")
    client.on_connect = on_connect
    client.on_message = on_message
    client.connect_async(MQTT_HOST, MQTT_PORT)  # broker chua len cung khong chet, tu ket noi lai
    client.loop_start()
    print(f"Slave 1 -> MQTT {MQTT_HOST}:{MQTT_PORT}")

    cua_so = CuaSoTrungBinh()
    nhip = time.monotonic()
    try:
        while True:
            nhiet_do, do_am = doc_dht()
            mau = {"nhiet_do": nhiet_do, "do_am": do_am, "bien_tro": doc_bien_tro(), "khoang_cach": doc_khoang_cach()}
            trang_thai = {ten: int(bong.is_lit) for ten, bong in led.items()}
            client.publish(TRANG_THAI, json.dumps(trang_thai), qos=1)
            trung_binh = cua_so.them(mau)
            if trung_binh:
                client.publish(TRUNG_BINH, json.dumps(trung_binh), qos=1)
                print("[GUI TRUNG BINH 20s]", trung_binh)
            print(mau, trang_thai)

            nhip += 1
            con_lai = nhip - time.monotonic()
            if con_lai > 0:
                time.sleep(con_lai)
            else:
                nhip = time.monotonic()  # DHT doc lai lau hon 1 s -> khong don nhip bu
    except KeyboardInterrupt:
        print("\nDa dung slave 1.")
    finally:
        for bong in led.values():
            bong.off()
        client.loop_stop()


if __name__ == "__main__":
    main()
```

- [ ] **Step 2: Viết `gateway/gateway.py`**

```python
"""
Buoi 11+12 - GATEWAY (pi4-tdbao): cau noi MQTT (master) <-> UDP (slave 2).

- Subscribe buoi11/dieu_khien -> gui lenh led3/buzzer/relay/light/alarm cho
  slave 2 qua UDP (cong 5005) ngay khi co lenh moi, va NHAC LAI moi 1 giay:
  UDP co the mat goi, lenh la TRANG THAI (0/1) nen gui lap lai khong sao.
- Nhan UDP tu slave 2 (cong 5006) {"loai": "trang_thai"|"trung_binh", ...}
  -> publish len buoi11/slave2/<loai> cho master.

Chay:
    IOT_MQTT_HOST=192.168.44.179 IOT_SLAVE2_HOST=192.168.44.176 python3 gateway.py
"""

import json
import os
import socket
import threading
from time import sleep

import paho.mqtt.client as mqtt

MQTT_HOST = os.environ.get("IOT_MQTT_HOST", "192.168.44.179")
MQTT_PORT = int(os.environ.get("IOT_MQTT_PORT", "1883"))
SLAVE2_HOST = os.environ.get("IOT_SLAVE2_HOST", "192.168.44.176")
CONG_SLAVE2 = 5005  # slave 2 nghe lenh
CONG_GATEWAY = 5006  # gateway nghe du lieu slave 2
DIEU_KHIEN = "buoi11/dieu_khien"
LENH_SLAVE2 = ("led3", "buzzer", "relay", "light", "alarm")
LOAI_HOP_LE = ("trang_thai", "trung_binh")

lenh = {}  # rong cho toi khi master gui lenh dau tien
khoa = threading.Lock()
udp = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
udp.bind(("0.0.0.0", CONG_GATEWAY))


def gui_lenh():
    with khoa:
        if not lenh:
            return
        goi = json.dumps(lenh).encode()
    try:
        udp.sendto(goi, (SLAVE2_HOST, CONG_SLAVE2))
    except OSError as loi:
        print("Khong gui duoc UDP toi slave 2:", loi)


def on_connect(client, userdata, connect_flags, reason_code, properties=None):
    print("Da ket noi MQTT:", reason_code)
    client.subscribe(DIEU_KHIEN, qos=1)


def on_message(client, userdata, msg):
    try:
        du_lieu = json.loads(msg.payload)
        with khoa:
            lenh.update({ten: int(du_lieu[ten]) for ten in LENH_SLAVE2 if ten in du_lieu})
    except Exception as loi:
        print("Lenh hong:", loi, msg.payload)
        return
    gui_lenh()
    print("Lenh moi cho slave 2:", lenh)


def vong_nhac_lenh():
    while True:
        gui_lenh()
        sleep(1)


def main():
    client = mqtt.Client(mqtt.CallbackAPIVersion.VERSION2, client_id="gateway-buoi11")
    client.on_connect = on_connect
    client.on_message = on_message
    client.connect_async(MQTT_HOST, MQTT_PORT)
    client.loop_start()
    threading.Thread(target=vong_nhac_lenh, daemon=True).start()
    print(f"Gateway: MQTT {MQTT_HOST}:{MQTT_PORT} | slave 2 {SLAVE2_HOST}:{CONG_SLAVE2} | nghe UDP :{CONG_GATEWAY}")
    try:
        while True:
            goi, dia_chi = udp.recvfrom(2048)
            try:
                du_lieu = json.loads(goi)
                loai = du_lieu.pop("loai")
                if loai not in LOAI_HOP_LE:
                    raise ValueError(f"loai la: {loai}")
            except Exception as loi:
                print("Goi UDP hong tu", dia_chi, loi)
                continue
            client.publish(f"buoi11/slave2/{loai}", json.dumps(du_lieu), qos=1)
            print(f"[slave2 -> {loai}]", du_lieu)
    except KeyboardInterrupt:
        print("\nDa dung gateway.")
    finally:
        client.loop_stop()


if __name__ == "__main__":
    main()
```

```python
    LED 3              -> D18  (GPIO18)
    Buzzer             -> D24  (GPIO24)
    Relay              -> D26  (GPIO26)
    Cam bien anh sang  -> A0   (ADC kenh 0)
    LCD 16x2 JHD1802   -> I2C
    USB LoRa           -> cong USB (/dev/ttyUSB0)

Chay (cung thu muc voi trung_binh.py, IoT_Driver_USB.py):
    IOT_GATEWAY_HOST=192.168.44.211 python3 slave2.py
"""

import json
import os
import socket
import threading
import time

from gpiozero import LED, Buzzer, OutputDevice
from grove.adc import ADC
from grove.display.jhd1802 import JHD1802

from IoT_Driver_USB import LORA_USB
from trung_binh import CuaSoTrungBinh

GATEWAY_HOST = os.environ.get("IOT_GATEWAY_HOST", "192.168.44.211")
CONG_SLAVE2 = 5005
CONG_GATEWAY = 5006
LORA_COM = os.environ.get("IOT_LORA_COM", "/dev/ttyUSB0")
NODE = 3
REG_DEN = {"light": 9, "alarm": 8}  # giong buoi 10: reg 9 = Light, reg 8 = Alarm Light

thiet_bi = {"led3": LED(18), "buzzer": Buzzer(24), "relay": OutputDevice(26)}
adc = ADC(0x08)
khoa = threading.Lock()
node = {"light": None, "alarm": None, "lora_ok": False}  # lan doc LoRa gan nhat
muon_node = {"light": None, "alarm": None}  # lenh master muon cho node 3

try:
    lcd = JHD1802()
except OSError as loi:  # chua cam LCD -> van chay, chi khong hien thi
    print("Khong thay LCD:", loi)
    lcd = None


def doc_anh_sang():
    try:
        return adc.read(0)  # 0-999 (phan nghin cua 3.3 V), cang sang cang lon
    except Exception as loi:
        print("Loi doc anh sang:", loi)
        return None


def hien_lcd(trang_thai, anh_sang):
    if lcd is None:
        return
    dong1 = f"L3:{trang_thai['led3']} BZ:{trang_thai['buzzer']} RL:{trang_thai['relay']}"
    dong2 = f"Anh sang:{'--' if anh_sang is None else anh_sang}"
    try:
        lcd.setCursor(0, 0)
        lcd.write(dong1.ljust(16)[:16])
        lcd.setCursor(1, 0)
        lcd.write(dong2.ljust(16)[:16])
    except OSError as loi:
        print("Loi LCD:", loi)


def vong_nhan_lenh():
    sock = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
    sock.bind(("0.0.0.0", CONG_SLAVE2))
    while True:
        goi, _ = sock.recvfrom(1024)
        try:
            lenh = json.loads(goi)
            for ten, tb in thiet_bi.items():
                if ten in lenh:
                    tb.on() if lenh[ten] else tb.off()
            with khoa:
                for ten in REG_DEN:
                    if ten in lenh:
                        muon_node[ten] = int(lenh[ten])
        except Exception as loi:
            print("Lenh UDP hong:", loi, goi)


def vong_lora():
    try:
        lora = LORA_USB(COM=LORA_COM, address=0)
    except Exception as loi:
        print("Khong mo duoc USB LoRa:", loi)
        return
    da_ghi = {"light": None, "alarm": None}
    while True:
        with khoa:
            muon = dict(muon_node)
        for ten, reg in REG_DEN.items():
            if muon[ten] is not None and muon[ten] != da_ghi[ten]:
                lora.write_data(NODE, reg, muon[ten])
                da_ghi[ten] = muon[ten]
                time.sleep(0.3)
        for ten, reg in REG_DEN.items():
            lora.ser.reset_input_buffer()
            gia_tri = lora.read_data(NODE, reg)
            with khoa:
                node["lora_ok"] = gia_tri != -1
                if gia_tri != -1:
                    node[ten] = 1 if gia_tri else 0
            if gia_tri != -1 and muon[ten] is not None and (1 if gia_tri else 0) != muon[ten]:
                da_ghi[ten] = None  # node chua theo lenh (mat goi / node khoi dong lai) -> ghi lai
            time.sleep(0.3)


def main():
    threading.Thread(target=vong_nhan_lenh, daemon=True).start()
    threading.Thread(target=vong_lora, daemon=True).start()
    udp = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)

    def gui(goi):
        try:
            udp.sendto(json.dumps(goi).encode(), (GATEWAY_HOST, CONG_GATEWAY))
        except OSError as loi:
            print("Khong gui duoc toi gateway:", loi)

    if lcd:
        lcd.clear()
    print(f"Slave 2 -> gateway {GATEWAY_HOST}:{CONG_GATEWAY} | nghe lenh UDP :{CONG_SLAVE2} | LoRa {LORA_COM}")
    cua_so = CuaSoTrungBinh()
    nhip = time.monotonic()
    try:
        while True:
            anh_sang = doc_anh_sang()
            trang_thai = {ten: int(bool(tb.value)) for ten, tb in thiet_bi.items()}
            with khoa:
                trang_thai.update(node)
            gui({"loai": "trang_thai", **trang_thai})
            trung_binh = cua_so.them({"anh_sang": anh_sang})
            if trung_binh:
                gui({"loai": "trung_binh", **trung_binh})
                print("[GUI TRUNG BINH 20s]", trung_binh)
            hien_lcd(trang_thai, anh_sang)
            print(f"anh sang {anh_sang} | {trang_thai}")

            nhip += 1
            con_lai = nhip - time.monotonic()
            if con_lai > 0:
                time.sleep(con_lai)
            else:
                nhip = time.monotonic()
    except KeyboardInterrupt:
        print("\nDa dung slave 2.")
    finally:
        for tb in thiet_bi.values():
            tb.off()
        if lcd:
            try:
                lcd.clear()
            except OSError:
                pass


if __name__ == "__main__":
    main()
```
